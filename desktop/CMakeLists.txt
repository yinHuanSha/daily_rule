cmake_minimum_required(VERSION 3.20)
project(DailyRule VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTORCC ON)
if(MSVC)
    add_compile_options(/utf-8)
endif()
find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Widgets Network Concurrent)
find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Widgets Network Concurrent)

add_library(rule_store store.cpp cloud.cpp clouddiff.cpp)
target_link_libraries(rule_store PUBLIC Qt${QT_VERSION_MAJOR}::Core)
add_library(rule_pages pages.cpp treepage.cpp)
target_link_libraries(rule_pages PUBLIC rule_store Qt${QT_VERSION_MAJOR}::Widgets)
add_library(rule_cloud cloudpage.cpp githubcloud.cpp)
target_link_libraries(rule_cloud PUBLIC rule_store Qt${QT_VERSION_MAJOR}::Widgets Qt${QT_VERSION_MAJOR}::Network Qt${QT_VERSION_MAJOR}::Concurrent)
if(WIN32)
    target_link_libraries(rule_cloud PRIVATE crypt32)
endif()
add_executable(DailyRule WIN32 main.cpp resources.qrc emblems.qrc)
if(WIN32)
    enable_language(RC)
    target_sources(DailyRule PRIVATE app.rc)
endif()
target_link_libraries(DailyRule PRIVATE rule_pages rule_cloud)
include(CTest)
if(BUILD_TESTING)
    add_executable(cloud_test cloud_test.cpp)
    target_link_libraries(cloud_test PRIVATE rule_store)
    add_test(NAME native_cloud COMMAND cloud_test)
    add_executable(store_test store_test.cpp)
    target_link_libraries(store_test PRIVATE rule_store)
    add_test(NAME native_store COMMAND store_test)
    find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Test)
    add_executable(ui_test ui_test.cpp resources.qrc emblems.qrc)
    target_link_libraries(ui_test PRIVATE rule_pages Qt${QT_VERSION_MAJOR}::Test)
    add_test(NAME native_ui COMMAND ui_test)
    set_tests_properties(native_ui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
